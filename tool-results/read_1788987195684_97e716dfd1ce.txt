   107→function SouthFacade({ progressRef }: { progressRef: RefObject<number> }) {
   108→  const parts = useMemo(() => {
   109→    const gold = new THREE.MeshStandardMaterial({
   110→      color: GOLD,
   111→      metalness: 0.7,
   112→      roughness: 0.25,
   113→      emissive: "#6E5424",
   114→      emissiveIntensity: 0.3,
   115→    });
   116→    const geos = {
   117→      mullionV: new THREE.BoxGeometry(0.09, 3.0, 0.09),
   118→      mullionH: new THREE.BoxGeometry(14.15, 0.07, 0.07),
   119→      post: new THREE.BoxGeometry(0.1, 3.02, 0.16),
   120→      header: new THREE.BoxGeometry(1.76, 0.14, 0.18),
   121→      handle: new THREE.CylinderGeometry(0.024, 0.024, 1.25, 10),
   122→      canopy: new THREE.BoxGeometry(2.4, 0.07, 0.55),
   123→    };
   124→    return { gold, geos };
   125→  }, []);
   126→
   127→  useFrame(() => {
   128→    const p = progressRef.current;
   129→    const fade = THREE.MathUtils.clamp((p - 0.46) / 0.08, 0, 1);
   130→    FACADE_GLASS_MAT.opacity = 0.32 * (1 - fade);
   131→    FACADE_GLASS_MAT.visible = fade < 0.999;
   132→  });
   133→
   134→  return (
   135→    <group>
   136→      <mesh material={FACADE_GLASS_MAT} position={[0, 1.85, 10]}>
   137→        <planeGeometry args={[14, 3.2]} />
   138→      </mesh>
   139→      {/* Vertical mullions every 2 m (center bay left open for the door). */}
   140→      {MULLION_XS.map((x) => (
   141→        <mesh
   142→          key={`mv-${x}`}
   143→          geometry={parts.geos.mullionV}
   144→          material={parts.gold}
   145→          position={[x, 1.75, 10.03]}
   146→        />
   147→      ))}
   148→      <mesh
   149→        geometry={parts.geos.mullionH}
   150→        material={parts.gold}
   151→        position={[0, 1.1, 10.03]}
   152→      />
   153→      <mesh
   154→        geometry={parts.geos.mullionH}
